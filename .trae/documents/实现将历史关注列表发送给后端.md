## 代码实现

### 修改 `platformApiService.ts`
```typescript
// 添加发送关注列表方法
public async sendFollowList(
  followedStreamers: FollowedStreamer[]
): Promise<void> {
  try {
    // 转换数据格式
    const watchConfig = {
      streamers: followedStreamers.map(streamer => ({
        platform: streamer.platform,
        id: streamer.id,
        nickname: streamer.nickname
      })),
      interval_ms: 5000,
      enable_notification: true
    };
    
    // 调用后端API
    await invoke<void>('start_follow_watch_service', {
      config: watchConfig
    });
    
    console.log('Successfully sent followed streamers to backend');
  } catch (error) {
    console.error('Failed to send followed streamers to backend:', error);
    throw error;
  }
}
```
