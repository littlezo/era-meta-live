- **功能**: 增加认证相关方法和增强现有方法
- **实现**: 
  ```rust

      }
      
      async fn login_with_code(
          &self,
          code: &str,
      ) -> Result<ApiResponse<()>, PlatformError> {
          Err(PlatformError::Unsupported("login_with_code not implemented".to_string()))
      }
      
      async fn refresh_token(
          &self,
      ) -> Result<ApiResponse<()>, PlatformError> {
          Err(PlatformError::Unsupported("refresh_token not implemented".to_string()))
      }
      
      async fn get_current_user(
          &self,
      ) -> Result<ApiResponse<serde_json::Value>, PlatformError> {
          Err(PlatformError::Unsupported("get_current_user not implemented".to_string()))
      }
      
      // 增强现有方法，支持返回原始数据
      async fn fetch_room_info_with_raw(
          &self,
          room_id: &str,
          with_raw: bool,
      ) -> Result<ApiResponse<RoomInfo>, PlatformError> {
          // 默认实现，调用原有方法并包装成ApiResponse

              data: room_info,
              raw: None,
              success: true,
              error: None,
              timestamp: chrono::Utc::now().timestamp_millis() as u64,
          })
      }
      
      // 其他方法类似增强...
  }
  ```
